// NovaActAPIBridge.js
export class NovaActAPIBridge {
  /**
   * Creates a new NovaAct API Bridge
   * @param {Object} options - Configuration options
   * @param {string} options.baseURL - Base URL for the API endpoints
   * @param {number} options.timeout - Request timeout in milliseconds
   * @param {string} options.apiKey - Optional API key for authentication
   */
  constructor(options = {}) {
    this.baseURL = options.baseURL || '/api/nova-act';
    this.timeout = options.timeout || 30000;
    this.apiKey = options.apiKey;
    this.sessionId = null;
    this.abortControllers = new Map();
  }

  /**
   * Creates a fetch request with consistent headers and timeout handling
   * @private
   */
  async _fetchWithTimeout(endpoint, options = {}) {
    const url = `${this.baseURL}${endpoint}`;
    const abortController = new AbortController();
    const id = Date.now().toString();
    this.abortControllers.set(id, abortController);

    const headers = {
      'Content-Type': 'application/json',
      ...(this.apiKey && { 'X-API-Key': this.apiKey }),
      ...(this.sessionId && { 'X-Session-ID': this.sessionId }),
      ...options.headers
    };

    const timeoutId = setTimeout(() => abortController.abort(), this.timeout);

    try {
      const response = await fetch(url, {
        ...options,
        headers,
        signal: abortController.signal
      });

      if (!response.ok) {
        const errorData = await response.json().catch(() => ({ message: 'Unknown error' }));
        throw new Error(errorData.message || `Request failed with status ${response.status}`);
      }

      return await response.json();
    } catch (error) {
      if (error.name === 'AbortError') {
        throw new Error(`Request timed out after ${this.timeout}ms`);
      }
      throw error;
    } finally {
      clearTimeout(timeoutId);
      this.abortControllers.delete(id);
    }
  }

  /**
   * Initializes a new NovaAct session
   * @param {Object} config - Configuration for NovaAct
   * @returns {Promise<Object>} Session information
   */
  async initialize(config) {
    try {
      // Make sure we don't have tty in the config as it's not supported
      const safeConfig = { ...config };
      delete safeConfig.tty;
      
      const data = await this._fetchWithTimeout('/initialize', {
        method: 'POST',
        body: JSON.stringify(safeConfig)
      });
      
      this.sessionId = data.sessionId;
      return data;
    } catch (error) {
      console.error('NovaAct initialization failed:', error);
      throw new Error(`Failed to initialize NovaAct: ${error.message}`);
    }
  }

  /**
   * Performs a browser action
   * @param {string|Object} action - The action to perform
   * @returns {Promise<Object>} Action result
   */
  async performAction(action) {
    if (!this.sessionId) {
      throw new Error('NovaAct session not initialized. Call initialize() first.');
    }
    
    try {
      return await this._fetchWithTimeout('/action', {
        method: 'POST',
        body: JSON.stringify({ action })
      });
    } catch (error) {
      console.error('NovaAct action failed:', error);
      throw new Error(`Failed to perform action: ${error.message}`);
    }
  }

  /**
   * Retrieves the current browser state
   * @returns {Promise<Object>} Current state
   */
  async getState() {
    if (!this.sessionId) {
      throw new Error('NovaAct session not initialized. Call initialize() first.');
    }
    
    try {
      return await this._fetchWithTimeout('/state', {
        method: 'GET'
      });
    } catch (error) {
      console.error('Failed to get NovaAct state:', error);
      throw new Error(`Failed to get state: ${error.message}`);
    }
  }

   * Takes a screenshot of the current browser window
   * @returns {Promise<string>} Base64-encoded screenshot

      throw new Error('NovaAct session not initialized. Call initialize() first.');

        method: 'POST'
      });

    } catch (error) {
      console.error('Failed to take screenshot:', error);
      throw new Error(`Failed to take screenshot: ${error.message}`);
    }
  }

   */

      return { status: 'closed', message: 'No active session' };
    }

      const result = await this._fetchWithTimeout('/close', {
        method: 'POST'
      });
      this.sessionId = null;
      return result;
    } catch (error) {
      console.error('Failed to close NovaAct session:', error);
      throw new Error(`Failed to close session: ${error.message}`);
    } finally {
      // Ensure sessionId is cleared even if request fails
      this.sessionId = null;
    }
  }

   * Cancels all pending requests

        if session_id in sessions:
            del sessions[session_id]
        logger.info(f"Session {session_id} closed at {get_current_formatted_time()}")
        return jsonify({
            "status": "success",
            "message": "NovaAct session closed successfully",
            "closed_at": get_current_formatted_time()
        })

        logger.error(f"Error closing session: {str(e)}")
        logger.error(traceback.format_exc())
        try:
            if session and session["id"] in sessions:
                del sessions[session["id"]]
        except Exception:
            pass
        return jsonify({"status": "error", "message": f"Error closing session: {str(e)}"}), 500

# ----------------------
# Session Cleanup Task
# ----------------------
def cleanup_sessions():
    """Clean up sessions idle for more than 30 minutes."""
    expiration_seconds = 1800  # 30 minutes
    while True:
        try:
            current_time = time.time()
            expired_sessions = []
            for session_id, session in list(sessions.items()):
                idle_time = current_time - session["last_used"]
                if idle_time > expiration_seconds:
                    expired_sessions.append(session_id)
            for session_id in expired_sessions:
                try:
                    sessions[session_id]["instance"].close()
                    logger.info(f"Expired session {session_id} closed (idle for {expiration_seconds} seconds)")
                except Exception as e:
                    logger.error(f"Error closing expired session {session_id}: {str(e)}")
                finally:
                    sessions.pop(session_id, None)
            # Check every 60 seconds
            time.sleep(60)
        except Exception as e:
            logger.error(f"Error in cleanup_sessions task: {str(e)}")
            time.sleep(60)

# Start the session cleanup thread in the background.
cleanup_thread = threading.Thread(target=cleanup_sessions, daemon=True)
cleanup_thread.start()
